\begin{frame}{Invariant point attention}
  \vspace{0.6cm}
  \begin{center}
    {\Large $a_{ij} \;=\; \operatorname{softmax}_j \Bigl( \bm{q}_i^{\top} \bm{k}_j \;+\; b_{ij} \;-\; \sum_{p} \bigl\| T_i \circ \vec{\bm{q}}_{ip} - T_j \circ \vec{\bm{k}}_{jp} \bigr\|^2 \Bigr)$}
    \\[1.2cm]
    {\large Usual attention, pair bias, distances between 3D points}
    \\[3mm]
    {\large Points live in the frame $T_i = (\bm{R}_i, \bm{t}_i)$ of residue $i$}
    \\[5mm]
    {\large \alert{Invariant to global rotation and translation}}
  \end{center}
  \reffootnote{Supplementary Information, Algorithm 22. One head, scaling constants omitted.}
\end{frame}

          \draw[ETHRed, line width=1pt, densely dashed] (2.0,1.3) -- (2.4,0.3);
          \fill[black] (2.0,1.3) circle (2.8pt);
          \fill[ETHBlue] (2.4,0.3) circle (2.8pt);
          \coordinate (gap) at (2.2,0.8);
        \end{scope}
        \fill[ETHRed] (0,0) circle (2.2pt);
        \node[font=\small, text=ETHRed, anchor=west] at ([xshift=2.5mm]gap) {error};
        \node[font=\footnotesize, text=ETHRed] at (0.15,-0.4) {$k$};
      \end{scope}
      \node[font=\small] at (11.8,-1.75) {both, seen from frame $k$};
    \end{tikzpicture}
    \\[5mm]
    {$\mathrm{FAPE} \;=\; \frac{1}{Z}\,\operatorname*{mean}_{k,\,i}\; \min\Bigl( d_{\mathrm{clamp}},\; \bigl\| (\bm{R}_k, \bm{t}_k)^{-1} \bm{x}_i \;-\; (\bm{R}_k^{\mathrm{true}}, \bm{t}_k^{\mathrm{true}})^{-1} \bm{x}_i^{\mathrm{true}} \bigr\| \Bigr)$,
    \quad $Z = d_{\mathrm{clamp}} = 10$\,\AA}
    \\[5mm]
    {\large \alert{Rotation: no change. Mirror image: penalised.}}
  \end{center}
\end{frame}
