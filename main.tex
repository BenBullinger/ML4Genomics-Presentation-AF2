% Draft (Ben, 14 October): the network in the message-passing formalism of geometric deep
% learning (Bronstein, Bruna, Cohen, Veličković 2021; notation of Veličković's EEML 2026 tutorial).
\begin{frame}[t]{AlphaFold 2 is message passing on a geometric graph}
  \vspace{-3mm}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.66\textwidth}
      {$\displaystyle \mathbf h_u=\phi\Big(\mathbf x_u,\ \bigoplus_{v\in\mathcal N_u}\psi(\mathbf x_u,\mathbf x_v,\mathbf e_{uv})\Big)$\par}
      \vspace{1mm}
      {\footnotesize\color{ETHGray} message $\psi$, aggregation $\bigoplus$, update $\phi$ (Bronstein et al.\ 2021). Nodes: residues, $\mathbf s_u$ and frame $T_u$. Edges: $\mathbf z_{uv}$.\par}
    \end{column}
    \begin{column}{0.3\textwidth}
      \centering
      \begin{tikzpicture}[x=0.6mm, y=0.6mm]
        \foreach \n/\pos/\ang in {u/{(0,0)}/60, v/{(20,5)}/-20, w/{(11,-12)}/110} {
          \draw[ETHRed, line width=1.1pt, -{Latex[length=3.5pt]}] \pos -- ++(\ang:4.5);
          \draw[ETHRed, line width=1.1pt, -{Latex[length=3.5pt]}] \pos -- ++(\ang+90:4.5);
          \fill[ETHBlue] \pos circle (2.2pt);
        }
        \node[text=ETHBlue, anchor=east] at (-2,0) {$u$};
        \node[text=ETHBlue, anchor=west] at (22,5) {$v$};
        \node[text=ETHBlue, anchor=north] at (11,-14.5) {$w$};
        \draw[black!50, line width=0.8pt, -{Latex[length=3.5pt]}] (2.5,0.6) -- (17.5,4.4) node[midway, above, sloped, font=\scriptsize, text=black!60] {$\mathbf z_{uv}$};
        \draw[black!50, line width=0.8pt, -{Latex[length=3.5pt]}] (1.5,-1.8) -- (9.5,-10.3);
        \draw[black!50, line width=0.8pt, -{Latex[length=3.5pt]}] (12.5,-10) -- (18.5,3);
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \vspace{1mm}
  {\scriptsize
  \renewcommand{\arraystretch}{1.0}
  \begin{tabular}{@{}lll@{}}
    \toprule
    Block & message $\psi$ & into \\
    \midrule
    MSA row attention & $\psi(\mathbf m_{su},\mathbf m_{sv},\mathbf z_{uv})$, the edge biases the weight & node $u$ \\
    outer product mean & $\psi(\mathbf m_{\cdot u},\mathbf m_{\cdot v})$, averaged over the sequences & edge $uv$ \\
    triangle updates & $\psi(\mathbf z_{uw},\mathbf z_{wv})$ over every third node $w$ & edge $uv$ \\
    IPA & $\psi(\mathbf s_u,\mathbf s_v,\mathbf z_{uv},\|T_u\circ\vec{\mathbf q}_u-T_v\circ\vec{\mathbf k}_v\|)$, values moved by $T_u^{-1}\!\circ T_v$ & node $u$ \\
    backbone update & none, $\phi$ alone & frame $T_u$ \\
    \bottomrule
  \end{tabular}\par}
  \vspace{0.5mm}
  \say{Triangles are two-hop messages. They make the edges consistent with one structure.}
  \key{Geometry enters only through distances. A frame is a gauge fixed by its atoms.}
\end{frame}
