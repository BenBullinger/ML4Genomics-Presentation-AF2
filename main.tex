\begin{frame}[t]{Backup (1/5): Exact $\mathrm{SE}(3)$ invariance of IPA}{SI 1.8.2, Algorithm 22, equations 3 to 6}
  \vspace{-3.5mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.26\textwidth}
      \vspace{2mm}
      \begin{tikzpicture}[x=3.8mm, y=3.8mm] \ipapair \end{tikzpicture}
    \end{column}
    \begin{column}{0.73\textwidth}
      \footnotesize
      \setlength{\abovedisplayskip}{1pt}\setlength{\belowdisplayskip}{1pt}
      {\raggedright Attention affinity uses pairwise Euclidean distances between query and key points:\par}
      \vspace{-1mm}
      \begin{align*}
        &\|(\Tg\circ T_i)\circ\vec{\bm q}_i-(\Tg\circ T_j)\circ\vec{\bm k}_j\|^2\\
        &\quad= \|\Tg\circ(T_i\circ\vec{\bm q}_i-T_j\circ\vec{\bm k}_j)\|^2
        = \|T_i\circ\vec{\bm q}_i-T_j\circ\vec{\bm k}_j\|^2
      \end{align*}
      {\color{ETHGray} Global rigid motion $\Tg \in \mathrm{SE}(3)$ cancels identically under Euclidean norm.\par}
      \vspace{1.5mm}
      {\raggedright Value aggregation mapped back into residue $i$ local frame:\par}
      \vspace{-1mm}
      \begin{align*}
        &(\Tg\circ T_i)^{-1}\circ\textstyle\sum_j a_{ij}\,(\Tg\circ T_j)\circ\vec{\bm v}_j\\
        &\quad= T_i^{-1}\circ\Tg^{-1}\circ\Tg\circ\textstyle\sum_j a_{ij}\,T_j\circ\vec{\bm v}_j
        = T_i^{-1}\circ\textstyle\sum_j a_{ij}\,T_j\circ\vec{\bm v}_j
      \end{align*}
      {\color{ETHGray} $\Tg^{-1}\Tg = I$ cancels algebraically; local representation is strictly invariant.\par}
    \end{column}
  \end{columns}
  \par\vspace{1mm}
  \key{Exact algebraic $\mathrm{SE}(3)$-invariance: attention weights and coordinate updates are independent of global pose.}
\end{frame}

\begin{frame}[t]{Backup (2/5): Stereochemical chirality and FAPE metric properties}{SI 1.8.1, 1.8.4, 1.9.4, Algorithm 28}
  \vspace{-3mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.33\textwidth}

    \begin{column}{0.65\textwidth}
      \footnotesize
      \setlength{\abovedisplayskip}{1.5pt}\setlength{\belowdisplayskip}{1.5pt}
      \say{Right-handed frames: Gram--Schmidt ($\bm e_3 = \bm e_1 \times \bm e_2$). Reflection negates local $z$:}
      \vspace{-1.5mm}
      \begin{equation*}
        T_i^{-1}\circ \bm x_j \mapsto \mathrm{diag}(1,1,-1)(T_i^{-1}\circ \bm x_j)
      \end{equation*}
      \say{$\mathrm{FAPE}(X, X_{\text{mirror}}) = \frac{2}{Z}\sum_{k,i} \min\big(d_{\text{clamp}}, |(T_k^{-1}\circ \bm x_i)_z|\big) > 0$.}
      {\color{ETHGray} Strictly positive for non-planar structures; breaks mirror reflection symmetry.\par}
      \vspace{0.8mm}
      \say{\textbf{Metric bound}: $\text{FAPE}(X, Y) \ge \min_{R,\bm t} \mathrm{RMSD}(R X + \bm t, Y)$. Non-negative, symmetric; satisfies triangle inequality (pseudometric on $\mathbb{R}^{3M}/\mathrm{SE}(3)$).}
    \end{column}
  \end{columns}
  \par\vspace{1.5mm}
  \key{Breaks reflection symmetry without Kabsch alignment while bounding optimal RMSD.}
\end{frame}
