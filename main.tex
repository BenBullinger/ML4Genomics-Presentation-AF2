\begin{frame}[t]{Backup (1/14): Why is IPA invariant to rigid motions?}{SI 1.8.2, Algorithm 22, equations 3 to 6}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.27\textwidth}
      \vspace{4mm}
      \begin{tikzpicture}[x=4mm, y=4mm] \ipapair \end{tikzpicture}
    \end{column}
    \begin{column}{0.72\textwidth}
      \small
      \setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}
      \say{The only geometry in the weights is the distance between points that $i$ and $j$ place in their own frames}
      \begin{align*}
        &\big\|(\Tg\circ T_i)\circ\vec{\bm q}_i-(\Tg\circ T_j)\circ\vec{\bm k}_j\big\|^2\\
        &\quad=\big\|\Tg\circ(T_i\circ\vec{\bm q}_i-T_j\circ\vec{\bm k}_j)\big\|^2
        =\big\|T_i\circ\vec{\bm q}_i-T_j\circ\vec{\bm k}_j\big\|^2
      \end{align*}
      {\color{ETHGray} moving both points together leaves their distance unchanged\par}
      \begin{align*}
        &(\Tg\circ T_i)^{-1}\circ\textstyle\sum_j a_{ij}\,(\Tg\circ T_j)\circ\vec{\bm v}_j\\
        &\quad=T_i^{-1}\circ\Tg^{-1}\circ\Tg\circ\textstyle\sum_j a_{ij}\,T_j\circ\vec{\bm v}_j
        =T_i^{-1}\circ\textstyle\sum_j a_{ij}\,T_j\circ\vec{\bm v}_j
      \end{align*}
      {\color{ETHGray} the output points are read back in frame $i$, and $\Tg$ meets its inverse. Exact, not learned.\par}
    \end{column}
  \end{columns}
  \key{Weights and outputs do not depend on where the structure sits}
\end{frame}

      \pictomirrorpage
    \end{column}
    \begin{column}{0.58\textwidth}
      \small
      \say{Residue frames from (N, C$\alpha$, C) via Gram--Schmidt ($\bm e_3=\bm e_1\times\bm e_2$): always right-handed.}
      \say{Reflecting all atoms leaves frames right-handed; only local $z$ flips:}
      \vspace{-1mm}
      {$T_i^{-1}\circ\bm x_j\ \mapsto\ \mathrm{diag}(1,1,-1)\,\big(T_i^{-1}\circ\bm x_j\big)$\par}
      \vspace{1.5mm}
      {$\mathrm{FAPE}(X,X_{\text{mirror}})=2\sum_{ij}\big|(T_i^{-1}\circ\bm x_j)_z\big|$\par}
      \vspace{1mm}
      {\color{ETHGray} Non-zero unless structure is strictly planar.\par}
      \vspace{2mm}
      \say{Models trained with reflection-blind dRMSD produce mirror images frequently (Suppl.\ Fig.~9).}
    \end{column}
  \end{columns}
  \key{FAPE breaks reflection symmetry; PDB training data teach L-amino acid chirality.}
\end{frame}
